\begin{definition}[Defense skill]
A \emph{defense skill} is an executable, auditable unit that maps an
intermediate agent observation to an allow / quarantine decision.
\end{definition}

\begin{theorem}[Soundness, demo]

\end{theorem}

\begin{algorithm}[t]
  \caption{Demo gatekeeper loop}
  \label{alg:demo}
  \begin{algorithmic}[1]
    \Require observation stream $o_1, o_2, \dots$
    \Ensure decision $d_t \in \{\text{allow}, \text{quarantine}\}$
    \ForAll{observations $o_t$ in the stream}
      \If{\textsc{Tier0Match}$(o_t)$}
        \State $d_t \gets \text{quarantine}$
      \Else
        \State $d_t \gets \textsc{Tier1Infer}(o_t)$
      \EndIf
    \EndFor
  \end{algorithmic}
\end{algorithm}
